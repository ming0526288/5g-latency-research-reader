python3 research/latency-analysis/tdd1/smoke_rfsim.py --configs research/latency-analysis/results/tdd1/rfsim-configs-01 --output research/latency-analysis/results/tdd1/rfsim-smoke-01 > research/latency-analysis/results/tdd1/preflight/smoke_matrix.txt 2>&1
